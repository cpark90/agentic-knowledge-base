PREFIX agt: <https://agentic-knowledge-base.dev/agt/>

PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
SELECT ?role ?access ?plane WHERE {
  ?role a agt:Role .
  { ?role agt:reads ?plane . BIND("reads" AS ?access) }
  UNION
  { ?role agt:writes ?plane . BIND("writes" AS ?access) }
  UNION
  { FILTER NOT EXISTS { ?role agt:reads ?any } BIND("reads" AS ?access) }
} ORDER BY ?role ?access ?plane
